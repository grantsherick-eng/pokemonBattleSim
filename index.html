var battle = createSprite(200, 200); battle.scale = 0.8; battle.setAnimation("battle"); var start = createSprite(200, 300); start.setAnimation("start"); var mewtwo = createSprite(300, 130); mewtwo.setAnimation("mewtwo"); mewtwo.scale = 0.3; mewtwo.visible = false; var slam = createSprite(264, 280); slam.setAnimation("bodyslam"); slam.visible = false; var rest = createSprite(370, 280); rest.setAnimation("rest"); rest.visible = false; var snore = createSprite(264, 320); snore.setAnimation("snore"); snore.visible = false; var yawn = createSprite(370, 320); yawn.setAnimation("yawn"); yawn.visible = false; var snorlax = createSprite(90, 250); snorlax.setAnimation("snorlax"); snorlax.scale = 0.5; snorlax.visible = false; var startActive = true; var canClick = true; var mewtwoHP = 100; var snorlaxHP = 100; // Turn Timer & Yawn Status var turn = "player"; // "player" or "mewtwo" var turnTimer = 0; var mewtwoDrowsy = false; var mewtwoAsleep = 0; function draw() { background("black"); if (mousePressedOver(start) && startActive) { battle.setAnimation("battle2"); startActive = false; start.visible = false; } if (startActive === false) { snorlax.visible = true; mewtwo.visible = true; slam.visible = true; rest.visible = true; snore.visible = true; yawn.visible = true; // PLAYER TURN if (turn === "player" && canClick && mewtwoHP > 0 && snorlaxHP > 0) { if (mousePressedOver(slam)) { mewtwoHP = mewtwoHP - 10; // Lowered from 12 startMewtwoDelay(); } if (mousePressedOver(rest)) { snorlaxHP = snorlaxHP + 25; // Lowered from 20 startMewtwoDelay(); } if (mousePressedOver(snore)) { mewtwoHP = mewtwoHP - 4; // Lowered from 8 startMewtwoDelay(); } if (mousePressedOver(yawn)) { if (mewtwoAsleep === 0) { mewtwoDrowsy = true; } startMewtwoDelay(); } } // MEWTWO TURN (1 second = 30 frames) if (turn === "mewtwo") { turnTimer = turnTimer - 1; if (turnTimer <= 0) { mewtwoTurn(); turn = "player"; } } } if (!mouseDown("leftButton")) { canClick = true; } if (mewtwoHP < 0) { mewtwoHP = 0; } if (mewtwoHP > 100) { mewtwoHP = 100; } if (snorlaxHP < 0) { snorlaxHP = 0; } if (snorlaxHP > 100) { snorlaxHP = 100; } drawSprites(); if (startActive === false) { fill("green"); rect(70, 112, mewtwoHP, 10); rect(300, 217, snorlaxHP, 10); } } function startMewtwoDelay() { canClick = false; turn = "mewtwo"; turnTimer = 30; // Back to 1 second } function mewtwoTurn() { if (mewtwoHP <= 0) { return; } // If Mewtwo is already asleep, it skips its turn if (mewtwoAsleep > 0) { mewtwoAsleep = mewtwoAsleep - 1; return; } // Mewtwo hits harder now var attack = randomNumber(1, 3); if (attack === 1) { snorlaxHP = snorlaxHP - 18; // Psychic (Increased from 10) } if (attack === 2) { snorlaxHP = snorlaxHP - 12; // Swift (Increased from 6) } if (attack === 3) { mewtwoHP = mewtwoHP + 15; // Recover (Increased from 10) } // At the END of the turn, drowsiness puts Mewtwo to sleep for 2 turns if (mewtwoDrowsy === true) { mewtwoDrowsy = false; mewtwoAsleep = 2; } }
